// CẤU HÌNH THÔNG TIN FILE GOOGLE SHEETS CHUẨN XÁC CỦA BẠN const FILE_SHEET_ID = '1JI8U96SwwQXn5ZLqWNlLtGaxkRzArpljjdW9QdIOr7A'; const SHEET_NAME = 'ALL'; function doGet(e) { if (e && e.parameter && e.parameter.action === 'getAll') { return json(getAllData()); } return json({ ok: true, message: 'API Quản lý học viên đang hoạt động' }); } function doPost(e) { try { const p = JSON.parse(e.postData.contents); if (p.action === 'getAll') { return json(getAllData()); } return json({ ok: false, message: 'Hành động không hợp lệ' }); } catch (err) { return json({ ok: false, message: 'Lỗi hệ thống: ' + err.message }); } } function getAllData() { const ss = SpreadsheetApp.openById(FILE_SHEET_ID); const sheet = ss.getSheetByName(SHEET_NAME); const data = sheet.getDataRange().getValues(); // Hàng số 2 trong file Trang tính chứa tiêu đề (Index 1 trong mảng) const headers = data[1]; const students = []; // Quét dữ liệu học viên bắt đầu từ Hàng số 3 trở đi (Index 2 trong mảng) for (let i = 2; i < data.length; i++) { const row = data[i]; if (!row || row.length === 0) continue; // Nếu cả cột Mã số HV (Cột C - index 2) và Họ tên (Cột D - index 3) trống thì bỏ qua if (!row[2] && !row[3]) continue; const student = {}; headers.forEach((header, index) => { if (header) { let value = row[index]; if (value instanceof Date) { value = Utilities.formatDate(value, Session.getScriptTimeZone(), "yyyy-MM-dd"); } student[header.toString().trim()] = value; } }); students.push(student); } return { students: students }; } function json(o) { return ContentService.createTextOutput(JSON.stringify(o)) .setMimeType(ContentService.MimeType.JSON); }
